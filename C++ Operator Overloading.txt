#include <iostream>
using namespace std;

class Number
{
  private:
    int val1;
    
    public:
        Number(int val)
        {
            val1 = val;
        }
        
        Number operator+ (Number& num)
        {
            return Number(val1 + num.val1);
        }
        Number operator- (Number& num)
        {
            return Number(val1 - num.val1);
        }
        Number operator* (Number& num)
        {
            return Number(val1 * num.val1);
        }
        Number operator/ (Number& num)
        {
            return Number(val1 / num.val1);
        }
        Number operator% (Number& num)
        {
            return Number(val1 % num.val1);
        }
        
        friend ostream& operator<<(ostream& COUT, Number& over_ld)
        {
            COUT << over_ld.val1 << endl;
            return COUT;
        }
};


int main()
{
    Number N1(5);
    Number N2(10);
    
    Number N3 = N1 + N2;
    cout << "Sum = " <<N3;
    
    N3 = N1 - N2;
    cout << "Diff = " <<N3;
    
    N3 = N1 * N2;
    cout << "Mul = " <<N3;
    
    N3 = N1 / N2;
    cout << "Div = " <<N3;
    
    N3 = N1 % N2;
    cout << "Mod = " <<N3;
    
    return 0;
}
