template <typename T>
class Calc
{
    public:
    T add (T a, T b)
    {
        return a + b;
    }
    
    T sub (T a, T b)
    {
        return a - b;
    }
    
    T mul (T a, T b)
    {
        return a * b;
    }
    
    T divide (T a, T b)
    {
        return a / b;
    }
};

int main()
{
    Calc<int> intCalc;
    
    cout << "Add = "<< intCalc.add(10,5)<< endl;

#include <iostream>
#include <list>
using namespace std;

void displayRating(const list<int>& rating)
{
    for(list<int>::const_iterator it = rating.begin(); it!=rating.end(); it++)
        cout << *it << endl;
}

void sortList(int newRating, list<int>& listName)
{
    for (list<int>::iterator it = listName.begin(); it!=listName.end(); it++)
    {
        if (*it >newRating)
        {
            listName.insert(it, newRating);
            return;
        }
    }
    listName.push_back(newRating);
    
}
int main()
{
    list<int>allPlayers = {2,5,7,9,1,4,6,8,1,3};
    list<int>beginners; 
    list<int>pros;

    for(list<int>::iterator it = allPlayers.begin(); it!=allPlayers.end(); it++)

        if ((*it >= 1) && (*it <=5))
            sortList(*it, beginners);

    }
    cout <<"Beginners :";
    displayRating(beginners);

**** FUNCTIONS AVAILABLE IN VECTOR ****

int main()
{
    vector<int> numbers;
    numbers.push_back(0);
    
    for (int i=1; i<10; i++)
        numbers.push_back(i);

    for (auto it=numbers.cbegin(); it != numbers.cend(); it++)
    {
        cout <<*it <<endl;
    }
        
    /* Size of the vector */    
    cout << "Size : " << numbers.size() << endl;

    /* The maximum possible number of elements the vector could theoretically hold. */   
    cout << "Max Size : " << numbers.max_size() << endl;

    /* Number of elements the vector can hold before needing to reallocate memory. */
    cout << "Capacity : " << numbers.capacity() << endl;

    /* Re-size of the vector */   
    numbers.resize(5);
    cout << "Size : " << numbers.size() << endl;
    for (auto it=numbers.cbegin(); it != numbers.cend(); it++)
    {
        cout <<*it <<endl;
    }
    
    /* Check whether vector is empty or not */   
    if (numbers.empty())
        cout << "Vertor is empty" << endl;
    else
        cout << "Vertor is not empty" << endl;
        
    /* Fetch element in a vector at given position */
    cout << "Element[2] : " << numbers[2] << endl;
    cout << "Element at (2) : " << numbers.at(2) << endl;

    /* To get the element which is present at front of the vector */	
    cout << "Numbers Front : " << numbers.front() << endl;
    
    /* To get the element which is present at back of the vector */
    cout << "Numbers Back : " << numbers.back() << endl;

    /* To clear the complete vector */
    numbers.clear();
    cout << "Size : " << numbers.size() << endl;

    return 0;
    
}

**** INSERTION AND DELETION OF ELEMENTS TO THE VECTOR ****

int main()
{
    vector<int> numbers;
    numbers.push_back(0);
    
    for (int i=1; i<10; i++)
        numbers.push_back(i);

    for (auto it=numbers.cbegin(); it != numbers.cend(); it++)
    {
        cout <<*it <<endl;
    }
        
        
    cout << "Size : " << numbers.size() << endl;
    
    
    /* Inserting element to the vector at the given postion */
    numbers.insert(numbers.begin()+5, 80);
    
    for (auto it=numbers.cbegin(); it != numbers.cend(); it++)
    {
        cout <<*it <<endl;
    }

    cout << "Size : " << numbers.size() << endl;

    
    /* Removing element in the vector at the given position */
    numbers.erase(numbers.begin()+5);
    for (auto it=numbers.cbegin(); it != numbers.cend(); it++)
    {
        cout <<*it <<endl;
    }
        
    cout << "Size : " << numbers.size() << endl;
    

    /* Removing element in the vector at the back */
    cout << "Removing element in the vector at the back" << endl;
    numbers.pop_back();
    for (auto it=numbers.cbegin(); it != numbers.cend(); it++)
    {
        cout <<*it <<endl;
    }
        
    cout << "Size : " << numbers.size() << endl;
    
    return 0;
}
